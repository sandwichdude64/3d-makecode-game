/*************************************************
Butterfly Migration: USA to Mexico

A 3D butterfly flying game built on the original 3D base engine.
Goal: fly from the USA to Mexico by collecting flowers and
maintaining enough energy to reach the coast.
*************************************************/

const settingPrefix = "butterflymigration_"

const USA_X = -90 << FP_BITS
const MEXICO_X = 90 << FP_BITS
const WORLD_START_Z = -40 << FP_BITS
const WORLD_END_Z = 40 << FP_BITS
const FLOWER_GOAL = 8

class ButterflyModel extends MeshTreeModelBase {
    static verticesFloat: number[][] = [
        // body
        [0, 0, 0],
        [0.35, 0.7, -0.7],
        [-0.35, 0.7, -0.7],
        [0.35, -0.7, -0.7],
        [-0.35, -0.7, -0.7],
        [0, 0.5, -2.4],

        // right wing upper
        [1.7, 0.8, -0.5],
        [3.2, 1.9, -1.2],
        [2.4, 0.8, -2.2],
        [1.1, 0.2, -1.0],

        // left wing upper
        [-1.7, 0.8, -0.5],
        [-3.2, 1.9, -1.2],
        [-2.4, 0.8, -2.2],
        [-1.1, 0.2, -1.0],

        // right wing lower
        [1.7, -0.8, -0.5],
        [3.2, -1.9, -1.2],
        [2.4, -0.8, -2.2],
        [1.1, -0.2, -1.0],

        // left wing lower
        [-1.7, -0.8, -0.5],
        [-3.2, -1.9, -1.2],
        [-2.4, -0.8, -2.2],
        [-1.1, -0.2, -1.0],
    ]

    static bodyFaces = [
        [0, 1, 2],
        [0, 1, 3],
        [0, 2, 4],
        [1, 3, 4, 2],
        [5, 1, 2],
    ]

    static rightWingUpperFaces = [ [6, 7, 8, 9] ]
    static leftWingUpperFaces = [ [10, 11, 12, 13] ]
    static rightWingLowerFaces = [ [14, 15, 16, 17] ]
    static leftWingLowerFaces = [ [18, 19, 20, 21] ]

    constructor() {
        super()
        const verticesFloat = ButterflyModel.verticesFloat
        const vertices = this.vertices
        const scale = 0.8

        for (let i = 0; i < verticesFloat.length; ++i) {
            const v = verticesFloat[i]
            vertices.push([
                Math.floor(v[0] * scale * FP_ONE),
                Math.floor(v[1] * scale * FP_ONE),
                Math.floor(v[2] * scale * FP_ONE)
            ])
        }

        const body = new MeshTreeNode(this, null, ButterflyModel.bodyFaces)
        const rwu = new MeshTreeNode(this, null, ButterflyModel.rightWingUpperFaces)
        const lwu = new MeshTreeNode(this, null, ButterflyModel.leftWingUpperFaces)
        const rwl = new MeshTreeNode(this, null, ButterflyModel.rightWingLowerFaces)
        const lwl = new MeshTreeNode(this, null, ButterflyModel.leftWingLowerFaces)

        body.addOutside(rwu)
        body.addOutside(lwu)
        body.addOutside(rwl)
        body.addOutside(lwl)

        this.meshTree = body
        this.calculateBoundingSphere()
        this.calculateNormalVectorsFromFaces()
    }
}

    static verticesFloat: number[][] = [
        [0, 0.9, 0],

        [0.5, -0.5, 0],
        [-0.5, -0.5, 0],

    ]

        [1, 2, 5],
        [2, 3, 5],

}

class TreeModel extends MeshTreeModelBase {
    static verticesFloat: number[][] = [
        // trunk
        [0, 0, 0],
        [0.5, 0, 0],
        [0, 4, 0],
        [0.5, 4, 0],
        // leaves
        [1, 5, 0],
        [-1, 5, 0],
        [0, 7, 0],
    ]

    static trunkFaces = [
        [0, 1, 3, 2],
        [0, 2, 6, 5],

class GroundModel extends MeshModelBase {
    constructor() {
        super()
        this.isTwoSided = true

        const baseY = -8 * FP_ONE
        const size = 120
        const step = 14

        for (let z = -12; z <= 12; ++z) {
            for (let x = -12; x <= 12; ++x) {
                const px = x * step
                const pz = z * step
                this.vertices.push([px * FP_ONE, baseY, -(pz * FP_ONE)])
                this.vertices.push([(px + step) * FP_ONE, baseY, -(pz * FP_ONE)])
                this.vertices.push([(px + step) * FP_ONE, baseY, -((pz + step) * FP_ONE)])
                this.vertices.push([px * FP_ONE, baseY, -((pz + step) * FP_ONE)])
                const p = this.vertices.length - 4
                this.faces.push([p, p + 1, p + 2, p + 3])
            }
        }

        this.calculateBoundingSphere()
        this.calculateNormalVectorsFromFaces()
    }
}

class ButterflyInstance extends InstanceBase {
    energy: number
    nectar: number
    reachedGoal: boolean

    constructor() {
        super()
        this.model = new ButterflyModel()
        this.energy = 100
        this.nectar = 0
        this.reachedGoal = false
    }

    updateWorldFromModel(multiplier: Fx8, shipMove: Fx8[], isSetupScreen: boolean) {
        this.setPosX(Fx.add(this.getPosX(), Fx.sub(Fx.mul(this.velocity[0], multiplier), shipMove[0])))
        this.setPosY(Fx.add(this.getPosY(), Fx.sub(Fx.mul(this.velocity[1], multiplier), shipMove[1])))
        this.setPosZ(Fx.add(this.getPosZ(), Fx.sub(Fx.mul(this.velocity[2], multiplier), shipMove[2])))

        this.energy -= 0.08
        if (this.energy < 0) this.energy = 0

        if (this.getPosX() > MEXICO_X) {
            this.reachedGoal = true
        }
    }
}

class FlowerInstance extends InstanceBase {
    collected: boolean
    glow: number

    constructor() {
        super()
        this.model = new FlowerModel()
        this.collected = false
        this.glow = Math.randomRange(0, 20)
    }

    updateWorldFromModel(multiplier: Fx8, shipMove: Fx8[], isSetupScreen: boolean) {
        if (this.collected) return

        // flower gently sways with the wind
        this.worldFromModel[10] += Fx8_to_FP(Fx8(0.02))
        this.worldFromModel[9] -= Fx8_to_FP(shipMove[0])
        this.worldFromModel[11] -= Fx8_to_FP(shipMove[2])
    }
}

class TreeInstance extends InstanceBase {
    constructor() {
        super()
        this.model = new TreeModel()
    }

    updateWorldFromModel(multiplier: Fx8, shipMove: Fx8[], isSetupScreen: boolean) {
        this.worldFromModel[9] -= Fx8_to_FP(shipMove[0])
        this.worldFromModel[11] -= Fx8_to_FP(shipMove[2])
    }
}

let butterfly: ButterflyInstance
let flowers: FlowerInstance[] = []
let trees: TreeInstance[] = []
let ground: GroundModel
let goalProgress = 0
let gameWon = false

function spawnFlowers() {
    flowers = []
    const total = FLOWER_GOAL + 4
    for (let i = 0; i < total; ++i) {
        const flower = new FlowerInstance()
        const angle = (i / total) * Math.PI * 2
        const dist = 15 + (i % 4) * 6
        flower.worldFromModel[9] = Math.floor(Math.cos(angle) * dist * FP_ONE)
        flower.worldFromModel[10] = Math.floor((-2 + (i % 3)) * FP_ONE)
        flower.worldFromModel[11] = Math.floor(Math.sin(angle) * dist * FP_ONE - 12 * FP_ONE)
        flowers.push(flower)
    }
}

function spawnTrees() {
    trees = []
    for (let i = 0; i < 9; ++i) {
        const tree = new TreeInstance()
        tree.worldFromModel[9] = Math.floor((i * 18 - 60) * FP_ONE)
        tree.worldFromModel[10] = Math.floor(-5 * FP_ONE)
        tree.worldFromModel[11] = Math.floor((Math.randomRange(-12, 12)) * FP_ONE)
        trees.push(tree)
    }
}

function resetButterflyFlight() {
    butterfly = new ButterflyInstance()
    butterfly.worldFromModel[9] = USA_X
    butterfly.worldFromModel[10] = 0
    butterfly.worldFromModel[11] = -10 << FP_BITS
    butterfly.velocity = [Fx.zeroFx8, Fx.zeroFx8, Fx.zeroFx8]

    ground = new GroundModel()
    spawnFlowers()
    spawnTrees()
    goalProgress = 0
    gameWon = false
}

function checkFlowerPickup() {
    for (let i = 0; i < flowers.length; ++i) {
        const flower = flowers[i]
        if (!flower || flower.collected) continue

        const dx = Fx.sub(butterfly.worldFromModel[9], flower.worldFromModel[9])
        const dy = Fx.sub(butterfly.worldFromModel[10], flower.worldFromModel[10])
        const dz = Fx.sub(butterfly.worldFromModel[11], flower.worldFromModel[11])
        const distSq = Fx.add(Fx.mul(dx, dx), Fx.add(Fx.mul(dy, dy), Fx.mul(dz, dz)))
        const pickupRadius = Fx8(4)

        if (Fx.compare(distSq, Fx.mul(pickupRadius, pickupRadius)) < 0) {
            flower.collected = true
            butterfly.nectar += 10
            butterfly.energy = Math.min(100, butterfly.energy + 18)
        }
    }
}

function updateMigrationProgress() {
    const dist = butterfly.getPosX() - USA_X
    const total = MEXICO_X - USA_X
    if (total <= 0) return
    goalProgress = Math.clamp(Math.floor((dist / total) * 100), 0, 100)
}

function updateButterflyControls(multiplier: Fx8) {
    const turnRate = Fx8(0.06)
    const liftRate = Fx8(0.05)
    const forwardSpeed = Fx8(0.2)

    if (controller.left.isPressed()) butterfly.roll -= turnRate
    if (controller.right.isPressed()) butterfly.roll += turnRate
    if (controller.up.isPressed()) butterfly.pitch += liftRate
    if (controller.down.isPressed()) butterfly.pitch -= liftRate

    butterfly.roll = Math.clamp(butterfly.roll as any as number, -0.7, 0.7) as any as Fx8
    butterfly.pitch = Math.clamp(butterfly.pitch as any as number, -0.7, 0.7) as any as Fx8

    butterfly.velocity[0] = Fx.mul(Fx8(1), forwardSpeed)
    butterfly.velocity[1] = Fx.mul(Fx8(0.25), butterfly.pitch)
    butterfly.velocity[2] = Fx.mul(Fx8(1), forwardSpeed)

    // world motion to the right as the butterfly flies toward Mexico
    butterfly.worldFromModel[9] += Fx8_to_FP(forwardSpeed)
}

game.onUpdate(function () {
    if (!butterfly || gameWon) return

    updateButterflyControls(Fx8(1))
    butterfly.updateWorldFromModel(Fx8(1), [Fx.zeroFx8, Fx.zeroFx8, Fx.zeroFx8], false)
    checkFlowerPickup()
    updateMigrationProgress()

    if (butterfly.energy <= 0) {
        game.splash("Out of energy! The butterfly needs more flowers.")
        resetButterflyFlight()
    }

    if (butterfly.nectar >= FLOWER_GOAL * 10 || butterfly.reachedGoal) {
        gameWon = true
        game.splash("You made it to Mexico! Nectar: " + butterfly.nectar)
    }
})
